## 5. 已实际运行的逐位assert

以下脚本本次已实际运行通过，既断言两列长度、每一位匹配与否，也断言兜的冲突位置，不是只比较首笔或笼统写“一致”。

```python
import re
from pathlib import Path

paths = {
    '兜': '/tmp/phz-full-visual/13/dou-guoxue.txt',
    '篼': '/tmp/phz-full-visual/dou-correction-篼-guoxue.txt',
    '蔸': '/tmp/phz-full-visual/dou-correction-蔸-guoxue.txt',
}
zdic = {
    '兜': '/tmp/phz-full-visual/13/dou-zdic.txt',
    '篼': '/tmp/phz-full-visual/10/篼-zdic.txt',
    '蔸': '/tmp/phz-full-visual/12/蔸-zdic.txt',
}
category = {
    '横': '1', '竖': '2', '撇': '3', '点': '4',
    '横折': '5', '竖提': '5', '竖弯钩': '5',
}
expected = {
    '兜': '32511355135',
    '篼': '31431432511355135',
    '蔸': '12232511355135',
}
expected_bad = {'兜': [2, 3, 4, 5, 6, 7], '篼': [], '蔸': []}
for ch, path in paths.items():
    m = re.search(r'^笔画\s+([1-5]+)\s+([^\n]+)',
                  Path(path).read_text(), re.M)
    digits, text = m.groups()
    names = text.split('、')
    assert digits == expected[ch]
    assert len(names) == len(digits)
    assert re.search(r'笔顺([1-5]+)',
                     Path(zdic[ch]).read_text()).group(1) == digits
    mapped = ''.join(category[n] for n in names)
    bad = [i for i, (d, n) in enumerate(zip(digits, names), 1)
           if d != category[n]]
    assert bad == expected_bad[ch]
    for i, (d, n) in enumerate(zip(digits, names), 1):
        assert (d == category[n]) == (i not in expected_bad[ch])
    assert (mapped == digits) == (ch != '兜')
assert expected['篼'][6:] == expected['蔸'][3:] == expected['兜']
assert expected['兜'][:5] == '32511'
assert ''.join(category[n] for n in ['撇', '竖', '横折', '横', '横']) == '32511'
assert category['竖提'] == '5'
assert '1' + '2' * 2 == '122'
```

结果：兜中文映射`35325115135`，冲突位`[2,3,4,5,6,7]`；篼17位、蔸14位全部一致。三份汉典数字字段均与对应汉语国学数字字段一致。
